## 场景描述

**场景**：李师傅（维修员，用户ID=3）给张三（学生，用户ID=4）发送聊天消息"你好"

**涉及角色**：
- 李师傅：消息发送者，维修员端用户
- 张三：消息接收者，学生端用户
- 后端服务：SpringBoot后端，处理消息存储和推送

```java
// 从令牌中解析用户ID
public Long getUserIdFromToken(String token) {
    // 验证令牌签名（HS256算法）
    Claims claims = Jwts.parser()
        .setSigningKey(SECRET_KEY)
        .parseClaimsJws(token)
        .getBody();
    
    // 从claims中获取用户ID
    return Long.parseLong(claims.getSubject());
}
```

```java
public static String toJson(Object obj) {
    ObjectMapper mapper = new ObjectMapper();
    // 注册Java 8时间模块
    mapper.registerModule(new JavaTimeModule());
    // 禁用日期序列化为时间戳
    mapper.disable(SerializationFeature.WRITE_DATES_AS_TIMESTAMPS);
    
    try {
        return mapper.writeValueAsString(obj);

        throw new RuntimeException("JSON序列化失败", e);

}
```

```java
// 第114-142行：sendMessage方法
public void sendMessage(Long userId, String message) {
    logger.info("=== WebSocket发送消息 ===");
    logger.info("目标用户ID: {}", userId);
    logger.info("当前在线用户数: {}", sessions.size());
    logger.info("在线用户列表: {}", sessions.keySet());
    
    // 从在线用户Map中获取张三的WebSocket会话
    WebSocketSession session = sessions.get(userId);
    // userId = 4（张三）
    
    if (session != null && session.isOpen()) {
        try {
            logger.info("用户 {} 在线，发送消息...", userId);
            
            // 将字符串包装成WebSocket消息对象
            TextMessage textMessage = new TextMessage(message);
            
            // 通过底层WebSocket连接发送数据
            session.sendMessage(textMessage);
            
            logger.info("WebSocket发送消息给用户 {} 成功: {}", userId, message);
        } catch (Exception e) {
            logger.error("WebSocket发送消息失败，用户ID: {}", userId, e);
        }
    } else {
        // 用户不在线
        logger.warn("WebSocket用户 {} 未在线（会话: {}, 状态: {}），消息将通过系统消息发送", 
            userId, session, session != null ? session.isOpen() : "null");
        
        // 消息已存入数据库，用户上线后可通过API查询
    }
}
```

```javascript
export function initWebSocket(userId, callback) {
  // 如果已有连接，先断开
  if (ws) {
    ws.close()
    ws = null
  }

  // 保存回调函数到模块级变量
  messageCallback = callback
  // messageCallback = handleWebSocketMessage

  // 构建WebSocket地址
  const backendHost = import.meta.env.VITE_WS_HOST || 'localhost:8080'
  const protocol = window.location.protocol === 'https:' ? 'wss:' : 'ws:'
  const url = `${protocol}//${backendHost}/ws/${userId}`
  // url = "ws://localhost:8080/ws/4"
  // 4是张三的用户ID

  // 创建WebSocket连接
  ws = new WebSocket(url)

  // 连接成功时触发
  ws.onopen = function() {
    console.log('WebSocket 连接成功, URL:', url)
    isConnected.value = true
  }

  // 收到消息时触发
  ws.onmessage = function(event) {
    console.log('=== 收到 WebSocket 消息 ===')
    console.log('原始消息:', event.data)
    
    try {
      // 解析JSON字符串为JavaScript对象
      const message = JSON.parse(event.data)
      // message = {
      //   id: 123,
      //   userId: 4,
      //   type: 2,
      //   subType: 3,
      //   title: "聊天消息",
      //   content: "你好",
      //   orderId: null,
      //   relatedUserId: 3,
      //   isRead: 0,
      //   createTime: "2026-05-09T10:30:00"
      // }
      
      console.log('解析后的消息:', message)
      
      // 调用保存的回调函数处理消息
      if (messageCallback) {
        console.log('调用消息回调...')
        messageCallback(message)
        // 执行 handleWebSocketMessage(message)
      }
    } catch (error) {
      console.error('解析 WebSocket 消息失败:', error)
    }
  }

  // 连接关闭时触发
  ws.onclose = function() {

    setTimeout(() => {
      initWebSocket(userId, callback)

    console.error('WebSocket 连接错误:', error)
```

## 五、事件分发阶段

### 5.1 eventBus发布事件

**文件**：`qianduan/src/utils/eventBus.js`

```javascript
// 第25-75行：eventBus对象
export const eventBus = {
  // 订阅事件
  on(eventName, callback) {
    if (!listeners.value[eventName]) {
      listeners.value[eventName] = []
    }
    listeners.value[eventName].push(callback)
  },

  // 发布事件
  emit(eventName, data) {
    if (listeners.value[eventName]) {
      // 遍历所有订阅者，调用他们的回调函数
      listeners.value[eventName].forEach(callback => {
        try {
          callback(data)
        } catch (error) {
          console.error('事件处理失败:', error)
        }
      })
    }
  },

  // 取消订阅
  off(eventName, callback) {
    if (!listeners.value[eventName]) return
    if (callback) {
      listeners.value[eventName] = listeners.value[eventName].filter(
        cb => cb !== callback
      )
    } else {
      listeners.value[eventName] = []
    }
  }
}
```

**事件发布流程**：

```javascript

  // 检查是否有订阅者
  if (listeners.value['message_received']) {
    // 遍历所有订阅了'message_received'事件的回调函数
    listeners.value['message_received'].forEach(callback => {
      try {
        // 调用每个订阅者的回调，传递消息数据
        callback(message)
      } catch (error) {
        console.error('事件处理失败:', error)
      }
    })
  }
}
```

## 十、技术要点总结

### 10.1 WebSocket连接管理

1. **连接建立**：用户登录后，布局组件挂载时自动建立连接
2. **连接地址**：`ws://localhost:8080/ws/{userId}`，包含用户ID
3. **自动重连**：连接断开后5秒自动重连
4. **连接状态**：通过`isConnected`响应式变量对外提供状态查询

### 10.2 消息持久化

1. **所有消息入库**：无论接收者是否在线，消息都存入数据库
2. **离线消息**：用户上线后通过API查询历史消息
3. **未读状态**：通过`is_read`字段标识，支持未读数统计

### 10.3 事件总线机制

1. **发布-订阅模式**：实现组件间解耦通信
2. **多订阅者支持**：一个事件可以有多个订阅者
3. **选择性处理**：订阅者根据消息内容决定是否更新UI
4. **内存管理**：组件卸载时取消订阅，防止内存泄漏

### 10.4 安全性保障

1. **JWT认证**：所有请求携带JWT令牌，验证用户身份
2. **用户隔离**：WebSocket连接包含用户ID，确保消息推送给正确用户
3. **权限控制**：只能查看与自己相关的消息

## 十二、常见问题解答

### Q1：如果接收者不在线怎么办？

**A**：消息会正常存入数据库，接收者上线后：
1. 通过API查询历史消息：`GET /api/notification/chat?otherUserId=2`
2. 通过未读数接口获取未读数量：`GET /api/notification/unread/count`
3. UI显示未读徽章，提醒用户有新消息

### Q2：如何确保消息只推送给正确的用户？

**A**：通过WebSocket连接地址包含用户ID实现：
```
ws://localhost:8080/ws/4  // 张三的连接
ws://localhost:8080/ws/2  // 李师傅的连接
```

后端通过`sessions.get(userId)`获取对应用户的WebSocket会话，确保消息只推送给正确的接收者。

### Q3：为什么需要eventBus？

**A**：eventBus实现组件间解耦通信：
1. WebSocket消息到达后，不知道哪些组件需要更新
2. 通过发布事件，所有订阅者都能收到通知
3. 组件根据自身需求选择性处理消息
4. 避免组件间直接依赖，提高可维护性

### Q4：消息会丢失吗？

**A**：不会丢失，原因：
1. 所有消息都存入数据库
2. WebSocket推送失败不影响消息存储
3. 用户上线后可查询历史消息
4. 未读状态持久化，支持离线提醒
